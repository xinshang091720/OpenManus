 public static class StaticRevitHttpServer
 {
     #region 静态全局私有字段
     private static ExternalEvent _globalApiEvent;
     private static readonly GlobalApiHandler _handler = new GlobalApiHandler();
     private static HttpListener _httpListener;
     private static Task _listenLoopTask;
     private static CancellationTokenSource _cts;
     // 可配置监听端口，默认5000，可外部修改

     public static int ListenPort = 5000;

     /// <summary>动态分配空闲端口</summary>
     private static int FindAvailablePort()
     {
         var listener = new TcpListener(IPAddress.Loopback, 0);
         listener.Start();
         var port = ((IPEndPoint)listener.LocalEndpoint).Port;
         listener.Stop();
         return port;
     }

     #endregion
     #region 对外静态启动/停止接口
     /// <summary>
     /// 启动HTTP监听服务（插件OnStartup调用）
     /// </summary>
     /// <returns>启动是否成功</returns>
     public static void StartListen()
     {
         try
         {
             // 初始化全局Revit事件（全局唯一）
             _globalApiEvent = ExternalEvent.Create(_handler);

             _cts = new CancellationTokenSource();
             _httpListener = new HttpListener();

             // 动态分配端口
             ListenPort = FindAvailablePort();
             string prefix = $"http://localhost:{ListenPort}/api/";
             _httpListener.Prefixes.Add(prefix);
             _httpListener.Start();

             // 启动成功后写入端口文件（供 WPF 端发现）
             var portDir = Path.Combine(Path.GetTempPath(), "BeeSync", "revit_ports");
             Directory.CreateDirectory(portDir);
             var portFile = Path.Combine(portDir, $"{Process.GetCurrentProcess().Id}.port");
             File.WriteAllText(portFile, ListenPort.ToString());

             // 启动监听循环后台任务
             _listenLoopTask = RunListenerLoop(_cts.Token);
         }
         catch (Exception ex)
         {
             // 日志记录
         }
     }

     /// <summary>
     /// 停止HTTP监听，释放全部资源（插件OnShutdown调用）
     /// </summary>
     public static void StopListen()
     {
         try
         {  // 清理端口文件
             var portFile = Path.Combine(Path.GetTempPath(), "BeeSync", "revit_ports",
                 $"{Process.GetCurrentProcess().Id}.port");
             try { File.Delete(portFile); } catch { }
             _cts?.Cancel();
             _httpListener?.Stop();
             _listenLoopTask?.Wait(3000);
             _httpListener?.Close();
             _globalApiEvent?.Dispose();
         }
         catch
         {
             // 忽略关闭阶段异常
         }
     }
     #endregion

     #region 持续接收HTTP请求循环（静态）
     private static async Task RunListenerLoop(CancellationToken token)
     {
         while (!token.IsCancellationRequested && _httpListener.IsListening)
         {
             HttpListenerContext ctx = null;
             try
             {
                 ctx = await _httpListener.GetContextAsync();
                 // 每一条请求新开任务处理，互不阻塞监听
                 _ = ProcessSingleRequest(ctx, token);
             }
             catch (HttpListenerException)
             {
                 // 正常Stop时抛出，直接退出循环
                 break;
             }
             catch (Exception ex)
             {
                 if (ctx != null)
                 {
                     await WriteError(ctx, 500, $"服务内部异常：{ex.Message}");
                 }
             }
         }
     }
     #endregion

     #region 处理单条POST请求（静态）
     private static async Task ProcessSingleRequest(HttpListenerContext ctx, CancellationToken token)
     {
         HttpListenerRequest req = ctx.Request;
         HttpListenerResponse resp = ctx.Response;
         try
         {
             string path = req.Url.AbsolutePath;
             string method = req.HttpMethod;
             string tokenHeader = req.Headers["Authorization"] ?? "";

             // 读取POST Body
             string bodyText = "";
             if (method == "POST")
             {
                 var reader = new StreamReader(req.InputStream);
                 bodyText = await reader.ReadToEndAsync();
             }
             // 只允许POST请求
             if (method != "POST")
             {
                 await WriteError(ctx, 405, "仅支持POST请求");
                 return;
             }
             // 封装本次请求所有参数
             var apiCtx = new ApiRequestContext
             {
                 ApiPath = path,
                 AuthToken = tokenHeader,
                 RequestBody = bodyText,
                 ResultTcs = new TaskCompletionSource<string>()
             };

             // 把当前请求上下文传给全局Handler
             _handler.SetRequestContext(apiCtx);
             // 唤醒Revit主线程执行对应接口逻辑
             _globalApiEvent.Raise();

             // 兼容.NET Framework等待逻辑
             var waitTask = Task.WhenAny(apiCtx.ResultTcs.Task, Task.Delay(-1, token));
             await waitTask;
             token.ThrowIfCancellationRequested();
             string resultJson = apiCtx.ResultTcs.Task.Result;

             // 返回业务JSON
             await WriteJsonResponse(resp, resultJson);
         }
         catch (TaskCanceledException)
         {
             await WriteError(ctx, 503, "Revit插件正在关闭，服务停止");
         }
         catch (Exception ex)
         {
             await WriteError(ctx, 500, $"请求处理失败：{ex.Message}");
         }
         finally
         {
             // 关键：无论正常/异常，强制关闭响应流释放socket资源
             resp?.OutputStream?.Close();
             resp?.Close();
         }
     }
     #endregion

     #region 通用返回工具静态方法
     private static async Task WriteJsonResponse(HttpListenerResponse resp, string json)
     {
         byte[] buffer = Encoding.UTF8.GetBytes(json);
         resp.ContentType = "application/json;charset=utf-8";
         resp.ContentLength64 = buffer.Length;
         await resp.OutputStream.WriteAsync(buffer, 0, buffer.Length);
     }

     private static async Task WriteError(HttpListenerContext ctx, int code, string msg)
     {
         ctx.Response.StatusCode = code;
         string errJson = $"{{\"code\":\"{code}\",\"msg\":\"{msg}\"}}";
         byte[] buf = Encoding.UTF8.GetBytes(errJson);
         ctx.Response.ContentType = "application/json;charset=utf-8";
         ctx.Response.ContentLength64 = buf.Length;
         await ctx.Response.OutputStream.WriteAsync(buf, 0, buf.Length);
         ctx.Response.OutputStream.Close();
     }
     #endregion
 }
